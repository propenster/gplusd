import random

import numpy as np
from sklearn import utils
import torch
from torch.utils.data import Dataset, DataLoader, random_split, ConcatDataset
from utils import reader, get_list_kmer, protein2num, neggen



def load_dataset(path, train_size=0.8, rand_neg=False, batch_size=32, num_cpu=4, device="cuda"):
    # Load the dataset from the given path
    # Split into training and testing sets based on train_size
    # If rand_neg is True, generate random negative samples
    # Return the training and testing datasets as PyTorch DataLoader objects
    manual_seed = torch.Generator().manual_seed(42)
    positive_samples = LoadEncoded(path, device=device)
    negative_samples = LoadEncoded(path, is_positive=False, fake=2, device=device)

    train_num = int(len(positive_samples) * train_size)
    val_num = int(len(positive_samples) * (1 - train_size)*0.5)
    split_size = [train_num, val_num, len(positive_samples) - train_num - val_num]
    train_pos, val_pos, test_pos = random_split(positive_samples, split_size, generator=manual_seed)
    train_neg, val_neg, test_neg = random_split(negative_samples, split_size, generator=manual_seed)

    if rand_neg:
        negative_data_rand = LoadEncoded(path, is_positive=False, fake=1, device=device)
        train_neg = ConcatDataset([train_neg, negative_data_rand])
    
    stack_datasets = [train_pos, val_pos, test_pos, train_neg, val_neg, test_neg]
    stack_loaders = [DataLoader(dataset, batch_size=batch_size, shuffle=True, num_workers=num_cpu) for dataset in stack_datasets]

    return stack_loaders


def load_data_test(path, batch_size=32, num_cpu=4, device="cuda"):
    dataset = LoadEncoded(path, device=device)
    return DataLoader(dataset, batch_size=batch_size, shuffle=False, num_workers=num_cpu)

class LoadEncoded(Dataset):
    def __init__(self, pathpos, is_positive=True, device="cuda", fake=0, length_pro=81, divide=5, part=2):
        if is_positive and fake != 0:
            raise Exception("Cant use key word fake on positive dataset")
        self.device = device
        self.fake = fake
        self.length_pro = length_pro
        self.divide = divide
        self.part = part

        # get list of kmer
        dic = get_list_kmer(1)

        # read data from file
        self.dpos = reader(pathpos)

        # convert protein to number sequence
        self.npos = [protein2num(pro, dic) for pro in self.dpos]

        if is_positive:
            self.poslabel = torch.from_numpy(np.ones(len(self.dpos)))
        else:
            # ic("go false")
            self.poslabel = torch.from_numpy(np.zeros(len(self.dpos)))
        self.poslabel = self.poslabel.to(device)

    def __len__(self):
        return len(self.dpos)

    def __getitem__(self, idx):
        # convert data to one hot format and up to device
        pro = self.npos[idx]
        if len(pro) < self.length_pro:
            pro = pro + [0] * (self.length_pro - len(pro))
        elif len(pro) > self.length_pro:
            pro = pro[:self.length_pro]

        # random generate a fake promoter by shuffle the pro
        if self.fake == 1:
            pro = random.shuffle(pro)
        # random generate a fake promoter by replace part of pro
        elif self.fake == 2:
            pro = neggen(pro, num_part=self.divide, keep=self.part, max_class=4)

        torchpro = torch.from_numpy(np.array(pro))
        onehot = torch.nn.functional.one_hot(torchpro, num_classes=4).to(self.device)
        return onehot.float(), self.poslabel[idx]

def extract_promoter_features(sequence, window_size=10):
    """
    Extracts tabular features that survive De Bruijn scrambling:
    1. Positional One-Hot 
    2. Local GC-Skew
    3. Thermodynamic Stability Proxy
    """
    sequence = sequence.upper()
    seq_len = len(sequence)
    features = {}
    
    # 1. Positional One-Hot Encoding
    mapping = {'A': [1,0,0,0], 'C': [0,1,0,0], 'G': [0,0,1,0], 'T': [0,0,0,1]}
    for i, base in enumerate(sequence):
        encoded = mapping.get(base, [0,0,0,0])
        features[f'pos_{i}_A'] = encoded[0]
        features[f'pos_{i}_C'] = encoded[1]
        features[f'pos_{i}_G'] = encoded[2]
        features[f'pos_{i}_T'] = encoded[3]
        
    # 2. Local GC-Skew & 3. Thermodynamic Stability Proxy
    stability_map = {'A': 2, 'T': 2, 'G': 3, 'C': 3}
    for i in range(seq_len - window_size + 1):
        window = sequence[i:i+window_size]
        
        # GC Skew
        g_count = window.count('G')
        c_count = window.count('C')
        skew = (g_count - c_count) / (g_count + c_count) if (g_count + c_count) > 0 else 0.0
        features[f'gc_skew_win_{i}'] = skew
        
        # Thermodynamic Stability (Hydrogen bonds)
        stability = sum(stability_map.get(base, 2.5) for base in window) / window_size
        features[f'stability_win_{i}'] = stability
        
    return features

def prepare_dataset(data_path, k_size=3):
    """Loads positive sequences, generates AEI negatives, and builds feature table."""
    print(f"Loading positive sequences from {data_path}...")
    with open(data_path, 'r') as f:
        positives = [line.strip() for line in f if line.strip()]

    print(f"Generating AEI negatives and extracting features...")
    data_rows = []
    
    for pos_seq in positives:
        # Positive Sample
        pos_features = extract_promoter_features(pos_seq)
        pos_features['label'] = 1
        data_rows.append(pos_features)
        
        # Negative Sample (De Bruijn Adversarial)
        neg_seq = generate_adversarial_negative(pos_seq, k=k_size)
        if not neg_seq: # Fallback if graph fails
            print("Fall back because graph fails...")
            neg_seq = pos_seq
        neg_features = extract_promoter_features(neg_seq)
        neg_features['label'] = 0
        data_rows.append(neg_features)
        
    return pd.DataFrame(data_rows)

import time
import pandas as pd
from Bio import SeqIO
from Bio.SeqRecord import SeqRecord

def transform(df):
    df = df.rename(columns={
        "Regulated Gene": "gene_id",
        "Absolute position": "coordinates_raw",
        "Binding seq.(cis-element)": "sequence"
    })
    data = []
    pos_coords = []
    
    for _, row in df.iterrows():
        coord_str = str(row["coordinates_raw"]).replace("..", "-")
        if pd.isna(row["sequence"]) or row["sequence"] in ["", "ND", "None"]:
            continue
            
        # Ensure we can split the coordinate string properly
        coords = coord_str.split("-")
        if len(coords) < 2:
            continue
            
        try:
            start, end = int(coords[0]), int(coords[-1])
            pos_coords.append((start, end))
            
            data.append({
                "gene_id": row["gene_id"],
                "gene_name": row["gene_id"],
                "coordinates": coord_str,
                "sequence": row["sequence"],
                "label": 1
            })
        except ValueError:
            # Skip rows where coordinates are malformed text instead of numbers
            continue
            
    return pd.DataFrame(data), pos_coords

def collect_dbtbs_data():
    sigmas = ["SigA", "SigB", "SigD", "SigE", "SigF", "SigG", "SigH", "SigK", "SigM", "SigW", "SigX", "SigY"]
    base_url = "https://dbtbs.hgc.jp/COG/tfac"
    all_sigma_data = []
    all_positive_coords = []
    good_cols = ["Regulated Gene", "Absolute position", "Binding seq.(cis-element)"]
    
    for sigma in sigmas:
        print(f"Running collect data for sigma {sigma}")
        try:
            df = pd.read_html(f"{base_url}/{sigma}.html")
            data_table = max(df, key=len)
            
            if not all(col in data_table.columns for col in good_cols):
                print(f"  Skipping {sigma}: missing required columns.")
                continue
                
            data_table = data_table[good_cols].copy()
            clean_df, pos_coords = transform(data_table)
            
            # Track which sigma factor this is for FASTA headers
            clean_df["sigma"] = sigma 
            
            all_sigma_data.append(clean_df)
            all_positive_coords.extend(pos_coords)
            
        except Exception as e:
            print(f"  Error while loading data for {sigma}: {e}")
        
        time.sleep(1)
    
    master_df = pd.concat(all_sigma_data, ignore_index=True)
    print(f"\nSuccessfully collected {len(master_df)} valid operons/sequences.")
    return master_df, all_positive_coords

def extract_dbtbs_promoters(promoter_df, pos_coords, genome_fasta, output_fasta):
    # Load the B. subtilis reference genome
    genome_record = next(SeqIO.parse(genome_fasta, "fasta"))
    genome_seq = genome_record.seq
    extracted_records = []
    
    for idx, row in promoter_df.iterrows():
        if idx >= len(pos_coords):
            break
            
        gene_name = str(row['gene_id']).replace(" ", "_").replace("/", "-")
        sigma = row['sigma']
        
        # Use the extracted coordinates from the transform function
        start_pos, end_pos = pos_coords[idx]
        
        # Convert 1-based biological coordinate to 0-based Python index
        anchor_index = start_pos - 1
        
        try:
            if start_pos < end_pos:
                # Plus strand: -60 upstream, +20 downstream
                slice_start = anchor_index - 60
                slice_end = anchor_index + 21
                segment = genome_seq[slice_start:slice_end]
            else:
                # Minus strand (start > end): Reverse the upstream/downstream logic
                slice_start = anchor_index - 20
                slice_end = anchor_index + 61
                segment = genome_seq[slice_start:slice_end].reverse_complement()
                
            if len(segment) == 81 and "N" not in segment.upper():
                record = SeqRecord(
                    segment,
                    id=f"{gene_name}_{sigma}_{start_pos}",
                    description="DBTBS_benchmark_81bp"
                )
                extracted_records.append(record)
                
        except IndexError:
            continue
            
    SeqIO.write(extracted_records, output_fasta, "fasta")
    print(f"Successfully extracted {len(extracted_records)} formatted sequences to {output_fasta}")

def extract_correct_dbtbs_promoters(promoter_df, genome_fasta, output_fasta):
    genome_record = next(SeqIO.parse(genome_fasta, "fasta"))
    genome_seq = genome_record.seq
    extracted_records = []
    
    for idx, row in promoter_df.iterrows():
        gene_name = str(row['gene_id']).strip().replace(" ", "_").replace("/", "-")
        sigma = row['sigma']
        binding_seq = str(row['sequence']).upper()
        
        # 1. Parse coordinates
        coords_raw = str(row['coordinates']).replace("..", "-").split("-")
        try:
            pos1, pos2 = int(coords_raw[0]), int(coords_raw[-1])
        except ValueError:
            continue
            
        # 2. Extract the raw genomic locus safely (handle max..min backwards notation)
        p_min, p_max = min(pos1, pos2), max(pos1, pos2)
        locus_fwd = genome_seq[p_min-1 : p_max].upper()
        
        # 3. Deduce strand
        strand = '+'
        if binding_seq in locus_fwd.reverse_complement():
            strand = '-'
            
        # 4. Estimate true TSS position to center the binding site at index ~25 (-35 box)
        if strand == '+':
            # 5' end of binding site is p_min. Estimated TSS is ~35bp downstream.
            tss_estimate = p_min + 35  
            slice_start = (tss_estimate - 1) - 60
            slice_end = (tss_estimate - 1) + 21
            segment = genome_seq[slice_start:slice_end]
        else:
            # 5' end of binding site on minus strand is p_max. Estimated TSS is ~35bp downstream (minus direction).
            tss_estimate = p_max - 35
            slice_start = (tss_estimate - 1) - 20
            slice_end = (tss_estimate - 1) + 61
            segment = genome_seq[slice_start:slice_end].reverse_complement()
            
        if len(segment) == 81 and "N" not in segment.upper():
            rec = SeqRecord(
                segment,
                id=f"{gene_name}_{sigma}_{tss_estimate}",
                description="DBTBS_benchmark_81bp"
            )
            extracted_records.append(rec)
            
    SeqIO.write(extracted_records, output_fasta, "fasta")
    print(f"Successfully extracted {len(extracted_records)} correctly aligned sequences to {output_fasta}")

import argparse
import numpy as np
import torch
import torch.nn as nn
from pathlib import Path
from sklearn.metrics import roc_curve, roc_auc_score, matthews_corrcoef, precision_score, recall_score, confusion_matrix

# 1. Define the model blueprint
class CNN_BiLSTM(nn.Module):
    def __init__(self):
        super(CNN_BiLSTM, self).__init__()
        self.conv = nn.Conv1d(4, 128, 15)
        self.relu = nn.ReLU()
        self.pool = nn.MaxPool1d(5)
        self.dropout1 = nn.Dropout(0.2)
        self.lstm = nn.LSTM(128, 32, bidirectional=True, batch_first=True)
        self.dropout2 = nn.Dropout(0.5)
        self.fc = nn.Linear(64, 1)
        self.sigmoid = nn.Sigmoid()

    def forward(self, x):
        x = self.dropout1(self.pool(self.relu(self.conv(x))))
        x = x.transpose(1, 2)
        x, _ = self.lstm(x)
        return self.sigmoid(self.fc(self.dropout2(x[:, -1, :])))

# 2. Define the tensor encoder
def get_onehot_tensor(seqs):
    mapping = {'A': [1,0,0,0], 'C': [0,1,0,0], 'G': [0,0,1,0], 'T': [0,0,0,1]}
    tensors = [np.array([mapping.get(b, [0.25]*4) for b in s]).T for s in seqs]
    return torch.tensor(np.array(tensors), dtype=torch.float32)

def evaluate_external(pos_path, neg_path, model_dir):
    print(f"Loading model from {model_dir}...")
    
    # 3. Load the PyTorch model properly
    model = CNN_BiLSTM()
    model.load_state_dict(torch.load(Path(model_dir) / "best_model.pt", map_location=torch.device('cpu')))
    model.eval()
    
    # Load sequences
    with open(pos_path, 'r') as f:
        pos_seqs = [line.strip() for line in f if len(line.strip()) == 81]
    with open(neg_path, 'r') as f:
        neg_seqs = [line.strip() for line in f if len(line.strip()) == 81]
        
    print(f"Evaluating on {len(pos_seqs)} external positives and {len(neg_seqs)} external negatives.")
    
    # Build dataset
    all_seqs = pos_seqs + neg_seqs
    y_true = np.array([1]*len(pos_seqs) + [0]*len(neg_seqs))
    
    # 4. Convert sequences to tensors and predict
    X = get_onehot_tensor(all_seqs)
    with torch.no_grad():
        y_proba = model(X).squeeze().numpy()
    
    # Calculate Metrics
    auc = roc_auc_score(y_true, y_proba)
    
    # --- FIND OPTIMAL THRESHOLD (Youden's J Statistic) ---
    fpr, tpr, thresholds = roc_curve(y_true, y_proba)
    youden_j = tpr - fpr
    optimal_idx = np.argmax(youden_j)
    optimal_threshold = thresholds[optimal_idx]
    
    print(f"\nDefault 0.5 Threshold is too strict. Adjusting to Optimal Threshold: {optimal_threshold:.4f}")
    
    # Apply new threshold
    y_pred = (y_proba >= optimal_threshold).astype(int)
    
    mcc = matthews_corrcoef(y_true, y_pred)
    precision = precision_score(y_true, y_pred, zero_division=0)
    recall = recall_score(y_true, y_pred, zero_division=0)
    cm = confusion_matrix(y_true, y_pred)
    
    print("\n" + "="*50)
    print("EXTERNAL VALIDATION RESULTS (AEI-CNN+BiLSTM)")
    print("="*50)
    print(f"AUC:       {auc:.4f}")
    print(f"MCC:       {mcc:.4f}")
    print(f"Precision: {precision:.4f}")
    print(f"Recall:    {recall:.4f}")
    print("-" * 50)
    print("Confusion Matrix:")
    print(f"True Negatives:  {cm[0][0]} | False Positives: {cm[0][1]}")
    print(f"False Negatives: {cm[1][0]} | True Positives:  {cm[1][1]}")
    print("="*50)

def parse_fasta_sigmas(fasta_path):
    """Parse FASTA headers formatted like '>abnA_SigA_2950133 DBTBS_benchmark_81bp'."""
    records = list(SeqIO.parse(fasta_path, "fasta"))
    data = []
    
    for rec in records:
        # rec.id gets sequence identifier prior to spaces (e.g., abnA_SigA_2950133)
        header_id = rec.id.split()[0]
        parts = header_id.split('_')
        
        # Extract sigma factor (e.g., 'SigA', 'SigB', 'SigK')
        sigma = "Unknown"
        for part in parts:
            if re.match(r'^Sig[A-Z0-9]+$', part, re.IGNORECASE):
                sigma = part.capitalize()
                break
                
        data.append({
            "id": header_id,
            "sigma": sigma,
            "sequence": str(rec.seq).upper()
        })
        
    df = pd.DataFrame(data)
    df.to_csv("dbtbs_benchmark_parsed.csv", index=False)
    print(f"Parsed {len(df)} sequences across {df['sigma'].nunique()} sigma factors:")
    print(df['sigma'].value_counts().to_string())
    return df

def one_hot_encode(sequences, channel_first=True):
    """Convert DNA sequences to one-hot tensors for CNN inputs."""
    mapping = {'A': 0, 'C': 1, 'G': 2, 'T': 3}
    encoded_list = []
    
    for seq in sequences:
        arr = np.zeros((4, len(seq)), dtype=np.float32)
        for i, base in enumerate(seq):
            if base in mapping:
                arr[mapping[base], i] = 1.0
        if not channel_first:
            arr = arr.T  # (L, 4)
        encoded_list.append(arr)
        
    return np.array(encoded_list)


def get_onehot_tensor(seqs):
    """The exact tensor encoder used during GPlusD training."""
    mapping = {'A': [1,0,0,0], 'C': [0,1,0,0], 'G': [0,0,1,0], 'T': [0,0,0,1]}
    tensors = [np.array([mapping.get(b, [0.25]*4) for b in s]).T for s in seqs]
    return torch.tensor(np.array(tensors), dtype=torch.float32)

class ModelPredictor:
    """Unified predictor wrapper handling CNNs (PyTorch/TF) or feature-based models."""
    def __init__(self, model_dir):
        self.model_dir = Path(model_dir)
        self.model_type = None
        self.model = None
        self.feature_cols = None
        self._load_model()

    def _load_model(self):
        pt_path = self.model_dir / "best_model.pt"
        # Create the model structure first
        self.model = CNN_BiLSTM()
        # Pour the .pt weights into the structure
        self.model.load_state_dict(torch.load(pt_path, map_location=torch.device('cpu')))
        self.model.eval()
        self.model_type = "pytorch_cnn"
        print(f"Loaded PyTorch CNN model from {pt_path}")
        return

    def predict_proba(self, sequences):
        X = get_onehot_tensor(sequences)
        with torch.no_grad():

            # FIX: The model already applies sigmoid natively. Do not apply it twice.
            probs = out.squeeze().numpy()

def main(fasta_path, neg_path, model_dir, output_dir, min_samples=10):
    Path(output_dir).mkdir(parents=True, exist_ok=True)
    
    # Load Model Predictor
    predictor = ModelPredictor(model_dir)
    
    # Parse Benchmark Data
    pos_df = parse_fasta_sigmas(fasta_path)
    
    # Load Background Negatives
    with open(neg_path, 'r') as f:
        neg_seqs = [line.strip().upper() for line in f if len(line.strip()) == 81]
    
    print(f"Loaded {len(neg_seqs)} background negative sequences.")
    neg_probs = predictor.predict_proba(neg_seqs)
    
    # Group by Sigma Factor and Compute Metrics
    sigmas = pos_df['sigma'].value_counts()
    valid_sigmas = sigmas[sigmas >= min_samples].index.tolist()
    
    fig, (ax_roc, ax_bar) = plt.subplots(1, 2, figsize=(16, 6))
    sns.set_theme(style="whitegrid")
    
    metrics = []
    
    for sigma in valid_sigmas:
        sig_seqs = pos_df[pos_df['sigma'] == sigma]['sequence'].tolist()
        pos_probs = predictor.predict_proba(sig_seqs)
        
        # Combine positives for this sigma with background negatives
        y_true = np.concatenate([np.ones(len(pos_probs)), np.zeros(len(neg_probs))])
        y_score = np.concatenate([pos_probs, neg_probs])
        
        # Compute Metrics
        auc = roc_auc_score(y_true, y_score)
        sensitivity = np.mean(pos_probs >= 0.5)
        fpr, tpr, _ = roc_curve(y_true, y_score)
        
        metrics.append({
            "Sigma Factor": sigma,
            "Count": len(sig_seqs),
            "AUC": auc,
            "Sensitivity": sensitivity
        })
        
        # Plot ROC curve for each sigma factor
        ax_roc.plot(fpr, tpr, label=f"{sigma} (n={len(sig_seqs)}, AUC={auc:.3f})", lw=2)

    # Styling ROC Panel
    ax_roc.plot([0, 1], [0, 1], 'k--', lw=1, alpha=0.7)
    ax_roc.set_xlim([0.0, 1.0])
    ax_roc.set_ylim([0.0, 1.05])
    ax_roc.set_xlabel('False Positive Rate', fontweight='bold')
    ax_roc.set_ylabel('True Positive Rate', fontweight='bold')
    ax_roc.set_title('ROC Curves Stratified by Sigma Factor', fontweight='bold', fontsize=14)
    ax_roc.legend(loc="lower right", frameon=True)

    # Plot Performance Metrics Bar Chart
    metrics_df = pd.DataFrame(metrics)
    
    # CORRECTED: Use 'value_name' instead of 'value_value_name'
    melted_df = pd.melt(metrics_df, id_vars=['Sigma Factor', 'Count'], 
                        value_vars=['AUC', 'Sensitivity'], 
                        var_name='Metric', value_name='Score')
    
    melted_df['Sigma Label'] = melted_df.apply(lambda r: f"{r['Sigma Factor']}\n(n={r['Count']})", axis=1)
    
    # CORRECTED: y axis must match 'Score'
    sns.barplot(data=melted_df, x='Sigma Label', y='Score', hue='Metric', 
                palette=['#1f77b4', '#2ca02c'], ax=ax_bar)
    
    ax_bar.set_ylim([0.0, 1.05])
    ax_bar.set_ylabel('Score', fontweight='bold')
    ax_bar.set_xlabel('Sigma Factor', fontweight='bold')
    ax_bar.set_title('Model Performance Metrics', fontweight='bold', fontsize=14)
    ax_bar.legend(loc="lower right")

    plt.tight_layout()
    plot_file = Path(output_dir) / 'sigma_factor_performance.pdf'
    plt.savefig(plot_file, dpi=600)
    print(f"\nSaved performance evaluation figure to {plot_file}")
    
    # Output Table
    print("\nDetailed Performance Table:")
    print(metrics_df.to_string(index=False, float_format="%.4f"))

import os

from Bio import SeqIO

def fasta_to_txt(fasta_path, txt_path, max_samples=None):
    sequences = []
    # Make sure the data folder exists
    os.makedirs(os.path.dirname(txt_path), exist_ok=True)
    
    for record in SeqIO.parse(fasta_path, "fasta"):
        seq = str(record.seq).upper()
        # Keep sequences that are exactly 81bp and have no unknown 'N' bases
        if len(seq) == 81 and 'N' not in seq:
            sequences.append(seq)
            
    # If a max limit is set, randomly downsample to balance the dataset
    if max_samples is not None and len(sequences) > max_samples:
        sequences = random.sample(sequences, max_samples)
    
    with open(txt_path, "w") as f:
        for seq in sequences:
            f.write(f"{seq}\n")
    return len(sequences)

class CNN_BiLSTM(nn.Module):
    def __init__(self):
        super(CNN_BiLSTM, self).__init__()
        self.conv = nn.Conv1d(4, 128, 15)
        self.relu = nn.ReLU()
        self.pool = nn.MaxPool1d(5)
        self.dropout1 = nn.Dropout(0.2)
        self.lstm = nn.LSTM(128, 32, bidirectional=True, batch_first=True)
        self.dropout2 = nn.Dropout(0.5)
        self.fc = nn.Linear(64, 1)
        self.sigmoid = nn.Sigmoid()

    def forward(self, x):
        x = self.dropout1(self.pool(self.relu(self.conv(x))))
        x = x.transpose(1, 2)
        x, _ = self.lstm(x)
        return self.sigmoid(self.fc(self.dropout2(x[:, -1, :]))).squeeze()

def get_onehot_tensor(seqs):
    mapping = {'A': [1,0,0,0], 'C': [0,1,0,0], 'G': [0,0,1,0], 'T': [0,0,0,1]}
    tensors = [np.array([mapping.get(b, [0.25]*4) for b in s]).T for s in seqs]
    return torch.tensor(np.array(tensors), dtype=torch.float32)

generate_negative_sequences.py
import random
from collections import defaultdict

def build_de_bruijn_graph(sequence, k):
    """Builds a De Bruijn graph from a DNA sequence."""
    graph = defaultdict(list)
    in_degree = defaultdict(int)
    out_degree = defaultdict(int)

    # Create nodes (k-1 mers) and edges (k mers)
    for i in range(len(sequence) - k + 1):
        kmer = sequence[i:i+k]
        prefix = kmer[:-1]
        suffix = kmer[1:]
        
        graph[prefix].append(suffix)
        out_degree[prefix] += 1
        in_degree[suffix] += 1

    return graph, in_degree, out_degree

def get_start_node(graph, in_degree, out_degree):
    """Finds the starting node for the Eulerian path."""
    start_node = None
    for node in set(in_degree.keys()).union(set(out_degree.keys())):
        # In a directed graph, the start node of an Eulerian path has out_degree - in_degree == 1
        if out_degree[node] - in_degree[node] == 1:
            return node
            
    # If it's a perfect Eulerian circuit (balanced degrees), pick any node with outgoing edges
    if not start_node:
        start_node = list(graph.keys())[0]
    return start_node

def generate_adversarial_negative(sequence, k=3):
    """
    Generates a negative sequence with the exact same k-mer frequency 
    as the input sequence by finding an alternative Eulerian path.
    """
    graph, in_degree, out_degree = build_de_bruijn_graph(sequence, k)
    start_node = get_start_node(graph, in_degree, out_degree)

    # NOVEL STEP: Shuffle the outgoing edges at every node. 
    # This forces Hierholzer's algorithm to take alternative branches,
    # destroying long-range structure while preserving k-mer counts.
    for node in graph:
        random.shuffle(graph[node])

    # Hierholzer's algorithm to find an Eulerian path
    stack = [start_node]
    path = []

    while stack:
        u = stack[-1]
        if graph[u]:
            # If there are unexplored edges, follow one
            v = graph[u].pop()
            stack.append(v)
        else:
            # If stuck, backtrack and add to path
            path.append(stack.pop())

    # The path is constructed backwards, so reverse it
    path = path[::-1]

    # Reconstruct the string from the path of (k-1)-mers
    if not path:
        return ""
        
    adversarial_seq = path[0]
    for node in path[1:]:
        adversarial_seq += node[-1]

    return adversarial_seq

# 1. Define the model blueprint
class CNN_BiLSTM(nn.Module):
    def __init__(self):
        super(CNN_BiLSTM, self).__init__()
        self.conv = nn.Conv1d(4, 128, 15)
        self.relu = nn.ReLU()
        self.pool = nn.MaxPool1d(5)
        self.dropout1 = nn.Dropout(0.2)
        self.lstm = nn.LSTM(128, 32, bidirectional=True, batch_first=True)
        self.dropout2 = nn.Dropout(0.5)
        self.fc = nn.Linear(64, 1)
        self.sigmoid = nn.Sigmoid()

    def forward(self, x):
        x = self.dropout1(self.pool(self.relu(self.conv(x))))
        x = x.transpose(1, 2)
        x, _ = self.lstm(x)
        return self.sigmoid(self.fc(self.dropout2(x[:, -1, :])))

# 2. Define the tensor encoder
def get_onehot_tensor(seqs):
    mapping = {'A': [1,0,0,0], 'C': [0,1,0,0], 'G': [0,0,1,0], 'T': [0,0,0,1]}
    tensors = [np.array([mapping.get(b, [0.25]*4) for b in s]).T for s in seqs]
    return torch.tensor(np.array(tensors), dtype=torch.float32)

import torch
from torch import nn

class ParallelCNN(nn.Module):
    def __init__(self, para_ker, pool_kernel=6, drop=0.5):
        super().__init__()
        self.lseq = nn.ModuleList([
            nn.Sequential(
                nn.Conv1d(4, 4, kernel_size=k, padding="same"),
                nn.ReLU(),
                nn.MaxPool1d(pool_kernel),
                nn.Dropout(drop)
            ) for k in para_ker
        ])

    def forward(self, inputs):
        return torch.cat([seq(inputs) for seq in self.lseq], dim=1)


class BidirectionalLSTM(nn.Module):
    def __init__(self, input_size, hidden_size, output_size):
        super().__init__()
        self.rnn = nn.LSTM(input_size, hidden_size, bidirectional=True, batch_first=True)
        self.linear = nn.Linear(hidden_size * 2, output_size)

    def forward(self, inputs):
        self.rnn.flatten_parameters()
        recurrent, _ = self.rnn(inputs)
        return self.linear(recurrent)


class GPlusD(nn.Module):
    def __init__(self, para_ker, input_shape=(64, 81, 4), pool_kernel=6, drop=0.5):
        super().__init__()
        binode = len(para_ker) * 4

        self.pconv = ParallelCNN(para_ker, pool_kernel, drop)
        self.bilstm = BidirectionalLSTM(binode, binode, binode)
        self.flatten = nn.Flatten()
        
        flat_dim = self._get_feature_shape(input_shape)

        self.fc = nn.Sequential(
            nn.Linear(flat_dim, flat_dim),
            nn.ReLU(),
            nn.Linear(flat_dim, 2),
            nn.Softmax(dim=1)
        )

    def _get_feature_shape(self, shape):
        with torch.no_grad():
            x = torch.zeros(shape).permute(0, 2, 1)
            x = self.pconv(x).permute(0, 2, 1)
            x = self.bilstm(x)
            return self.flatten(x).shape[1]

    def forward(self, x):
        x = self.pconv(x.permute(0, 2, 1)).permute(0, 2, 1)
        x = self.bilstm(x)
        return self.fc(self.flatten(x))

def generate_external_dataset(pos_out="data/cnnprom_positive.txt", neg_out="data/cnnprom_negative.txt", seq_length=81):
    os.makedirs("data", exist_ok=True)
    
    print("Fetching Bacillus subtilis reference genome (NC_000964.3) directly from NCBI...")
    try:
        # Fetch the full annotated GenBank file
        handle = Entrez.efetch(db="nucleotide", id="NC_000964.3", rettype="gbwithparts", retmode="text")
        record = SeqIO.read(handle, "genbank")
        handle.close()
    except Exception as e:
        print(f"Failed to fetch from NCBI. Error: {e}")
        return

    genome_seq = record.seq
    positives = []
    
    print(f"Length of genome fetched from NCBI >>> {len(genome_seq)}")
    print("Extracting upstream regulatory regions (-60 to +20) for all CDS...")
    
    # Extract upstream regions of annotated genes as presumed true promoters
    for feature in record.features:
        if feature.type == "CDS":
            # FIXED: Access strand through feature.location
            if feature.location.strand == 1: # Forward strand
                tss = feature.location.start
                start = tss - 60
                end = tss + 21
                if start >= 0:
                    seq = str(genome_seq[start:end]).upper()
                    if len(seq) == seq_length and 'N' not in seq:
                        positives.append(seq)
            elif feature.location.strand == -1: # Reverse strand
                tss = feature.location.end
                start = tss - 21
                end = tss + 60
                if end <= len(genome_seq):
                    # Remember to reverse complement the reverse strand!
                    seq = str(genome_seq[start:end].reverse_complement()).upper()
                    if len(seq) == seq_length and 'N' not in seq:
                        positives.append(seq)
                        
    # Save Positives
    with open(pos_out, "w") as f:
        for seq in positives:
            f.write(f"{seq}\n")
    print(f"Saved {len(positives)} putative promoters to {pos_out}")

    # Generate Negatives
    print("Generating background genomic negatives...")
    negatives = []
    genome_str = str(genome_seq).upper()
    
    while len(negatives) < len(positives):
        start = random.randint(0, len(genome_str) - seq_length)
        chunk = genome_str[start:start+seq_length]
        if 'N' not in chunk:
            negatives.append(chunk)
            
    with open(neg_out, "w") as f:
        for seq in negatives:
            f.write(f"{seq}\n")
    print(f"Saved {len(negatives)} real genome background sequences to {neg_out}")
    print("External dataset preparation complete!")

def random_shuffle_seq(seq):
    """Method 1: Complete mononucleotide random shuffle."""
    seq_list = list(seq)
    random.shuffle(seq_list)
    return "".join(seq_list)

def block_shuffle_seq(seq, num_blocks=10, shuffle_ratio=0.6):
    """Method 2: DeePromoter-style block segmentation and partial shuffling."""
    block_size = len(seq) // num_blocks
    blocks = [seq[i*block_size:(i+1)*block_size] for i in range(num_blocks)]
    
    num_to_shuffle = int(num_blocks * shuffle_ratio)
    indices_to_shuffle = random.sample(range(num_blocks), num_to_shuffle)
    
    shuffled_sub_blocks = [blocks[i] for i in indices_to_shuffle]
    random.shuffle(shuffled_sub_blocks)
    
    for idx, new_block in zip(indices_to_shuffle, shuffled_sub_blocks):
        blocks[idx] = new_block
        
    return "".join(blocks)

# --- 1. DEEP LEARNING ARCHITECTURE (PyTorch) ---
class CNN_BiLSTM(nn.Module):
    def __init__(self):
        super(CNN_BiLSTM, self).__init__()
        self.conv = nn.Conv1d(in_channels=4, out_channels=128, kernel_size=15)
        self.relu = nn.ReLU()
        self.pool = nn.MaxPool1d(kernel_size=5)
        self.dropout1 = nn.Dropout(0.2)
        # Bidirectional LSTM: 128 input features, 32 hidden size -> 64 output size
        self.lstm = nn.LSTM(input_size=128, hidden_size=32, bidirectional=True, batch_first=True)
        self.dropout2 = nn.Dropout(0.5)
        self.fc = nn.Linear(64, 1)
        self.sigmoid = nn.Sigmoid()

    def forward(self, x):
        x = self.conv(x)
        x = self.relu(x)
        x = self.pool(x)
        x = self.dropout1(x)
        x = x.transpose(1, 2) # Swap dimensions for LSTM (batch, seq_len, features)
        x, _ = self.lstm(x)
        x = x[:, -1, :] # Take the output of the last time step
        x = self.dropout2(x)
        x = self.fc(x)
        return self.sigmoid(x).squeeze()

def get_onehot_tensor(seqs):
    mapping = {'A': [1,0,0,0], 'C': [0,1,0,0], 'G': [0,0,1,0], 'T': [0,0,0,1]}
    tensors = []
    for seq in seqs:
        encoded = [mapping.get(base, [0.25,0.25,0.25,0.25]) for base in seq]
        tensors.append(np.array(encoded).T) # Shape: (4, seq_len)
    return torch.tensor(np.array(tensors), dtype=torch.float32)

# --- 2. PIPELINE FUNCTIONS ---
def evaluate_model(y_true, y_proba, model_name):
    auc = roc_auc_score(y_true, y_proba)
    fpr, tpr, thresholds = roc_curve(y_true, y_proba)
    optimal_idx = np.argmax(tpr - fpr)
    optimal_threshold = thresholds[optimal_idx]
    
    y_pred = (y_proba >= optimal_threshold).astype(int)
    mcc = matthews_corrcoef(y_true, y_pred)
    precision = precision_score(y_true, y_pred, zero_division=0)
    recall = recall_score(y_true, y_pred, zero_division=0)
    
    return {'Model': model_name, 'Precision': precision, 'Recall': recall, 'MCC': mcc, 'AUC': auc}

def train_pytorch_model(model_name, train_seqs, y_train_labels, test_tensor, y_test, epochs=25):
    print(f"Training {model_name}...")
    X_train_tensor = get_onehot_tensor(train_seqs)
    y_train_tensor = torch.tensor(y_train_labels, dtype=torch.float32)
    
    dataset = TensorDataset(X_train_tensor, y_train_tensor)
    loader = DataLoader(dataset, batch_size=64, shuffle=True)
    
    model = CNN_BiLSTM()
    criterion = nn.BCELoss()
    optimizer = optim.AdamW(model.parameters(), lr=1e-3)
    
    model.train()
    for epoch in range(epochs): 
        for batch_X, batch_y in loader:
            optimizer.zero_grad()
            outputs = model(batch_X)
            loss = criterion(outputs, batch_y)
            loss.backward()
            optimizer.step()
            
    model.eval()
    with torch.no_grad():
        y_proba = model(test_tensor).numpy()
    return evaluate_model(y_test, y_proba, model_name)

def evaluate(net, loaders):
    eval_result = list()
    ltotal = list()
    lcorrect = list()
    pred_result = list()
    true_result = list()
    for load in loaders:
        total = 0
        correct = 0
        pred_list = list()
        true_list = list()
        for data in load:
            inputs = data[0]
            labels = data[1]
            outputs = net(inputs)
            _, predicted = torch.max(outputs.data, 1)
            total += labels.size(0)
            correct += (predicted == labels).sum().item()
            pred_list += list(predicted.cpu().numpy())
            true_list += list(labels.cpu().numpy())
        acc = correct / total
        eval_result.append(acc)
        lcorrect.append(correct)
        ltotal.append(total)
        pred_result.append(pred_list)
        true_result.append(true_list)
    return (lcorrect, ltotal), pred_result, true_result


def mcc(data):
    pos_count = data[0][0]
    neg_count = data[0][1]

    tol_pos_count = data[1][0]
    tol_neg_count = data[1][1]

    TP = pos_count
    FN = tol_pos_count - pos_count
    TN = neg_count
    FP = tol_neg_count - neg_count

    precision = TP / (TP + FP)
    recall = TP / (TP + FN)
    MCC = (TP * TN - FP * FN) / math.sqrt((TP + FP) * (TP + FN) * (TN + FP) * (TN + FN))

    return precision, recall, MCC

import argparse
from pathlib import Path

import torch
import torch.optim as optim
from torch import nn
from icecream import ic

from dataloader import load_dataset
from models import GPlusD
from test import evaluate, mcc

device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')


def train(data_path, pretrain=None, outputdir="test", training=True, ker=None, epoch_num=1000):
    ker = ker or [27, 14, 7]

    # Create experiment folders
    exp_folder = Path(f"./output/{outputdir}")
    exp_folder.mkdir(parents=True, exist_ok=True)

    ic("Data loading")
    train_pos, val_pos, test_pos, train_neg, val_neg, test_neg = load_dataset(data_path, device=device)

    net = GPlusD(ker).to(device)
    if pretrain:
        net.load_state_dict(torch.load(pretrain))

    criterion = nn.CrossEntropyLoss()
    optimizer = optim.Adam(net.parameters(), lr=0.00001)

    best_mcc, best_precision, best_recall = 0, 0, 0
    patience_counter = 0
    patience_limit = 10  # Stops after 10 evaluation cycles without MCC improvement

    if training:
        ic("Start training")
        for epoch in range(epoch_num):
            net.train()
            
            for (batch_pos, batch_neg) in zip(train_pos, train_neg):
                inputs = torch.cat((batch_pos[0], batch_neg[0]), dim=0)
                labels = torch.cat((batch_pos[1], batch_neg[1]), dim=0).long()

                optimizer.zero_grad()
                outputs = net(inputs)
                loss = criterion(outputs, labels)
                loss.backward()
                optimizer.step()

            # Evaluation cycle every 10 epochs
            if epoch % 10 == 0:
                torch.save(net.state_dict(), exp_folder / f"epoch_{epoch}.pth")
                
                net.eval()
                with torch.no_grad():
                    eval_data, _ = evaluate(net, [val_pos, val_neg])
                    precision, recall, MCC = mcc(eval_data)

                ic(epoch, outputdir, precision, recall, MCC)

                # Save best models
                if precision > best_precision:
                    best_precision = precision
                    ic("Update best precision")
                    torch.save(net.state_dict(), exp_folder / "best_precision.pth")
                    
                if recall > best_recall:
                    best_recall = recall
                    ic("Update best recall")
                    torch.save(net.state_dict(), exp_folder / "best_recall.pth")
                    
                if MCC > best_mcc:
                    best_mcc = MCC
                    ic("Update best MCC")
                    torch.save(net.state_dict(), exp_folder / "best_mcc.pth")
                    patience_counter = 0
                else:
                    patience_counter += 1

                # Early stopping
                if patience_counter >= patience_limit:
                    ic("Early stopping triggered")
                    break

    # Final Testing Phase
    ic("Start testing phase")
    net.load_state_dict(torch.load(exp_folder / "best_mcc.pth"))
    net.eval()
    
    with torch.no_grad():
        eval_data, _ = evaluate(net, [test_pos, test_neg])
        precision, recall, MCC = mcc(eval_data)

    ic(precision, recall, MCC)

    # Save test logs
    with open(exp_folder / "log.txt", "w") as f:
        f.write(f"Test precision: {precision}\nTest recall: {recall}\nTest MCC : {MCC}\n")


if __name__ == '__main__':

train_cnn_bilstm.py
import argparse
import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import TensorDataset, DataLoader
from sklearn.model_selection import StratifiedKFold
from sklearn.metrics import roc_auc_score, matthews_corrcoef
from pathlib import Path

class CNN_BiLSTM(nn.Module):
    def __init__(self):
        super(CNN_BiLSTM, self).__init__()
        self.conv = nn.Conv1d(4, 128, 15)
        self.relu = nn.ReLU()
        self.pool = nn.MaxPool1d(5)
        self.dropout1 = nn.Dropout(0.2)
        self.lstm = nn.LSTM(128, 32, bidirectional=True, batch_first=True)
        self.dropout2 = nn.Dropout(0.5)
        self.fc = nn.Linear(64, 1)
        self.sigmoid = nn.Sigmoid()

    def forward(self, x):
        x = self.dropout1(self.pool(self.relu(self.conv(x))))
        x = x.transpose(1, 2)
        x, _ = self.lstm(x)
        return self.sigmoid(self.fc(self.dropout2(x[:, -1, :]))).squeeze()

def get_onehot_tensor(seqs):
    mapping = {'A': [1,0,0,0], 'C': [0,1,0,0], 'G': [0,0,1,0], 'T': [0,0,0,1]}
    tensors = [np.array([mapping.get(b, [0.25]*4) for b in s]).T for s in seqs]
    return torch.tensor(np.array(tensors), dtype=torch.float32)

def train_pytorch(data_path, neg_path, output_dir):
    Path(output_dir).mkdir(parents=True, exist_ok=True)
    
    with open(data_path, 'r') as f:
        pos = [l.strip().upper() for l in f if len(l.strip()) == 81]
    with open(neg_path, 'r') as f:
        neg = [l.strip().upper() for l in f if len(l.strip()) == 81]
        
    X_tensor = get_onehot_tensor(pos + neg)
    y = np.array([1]*len(pos) + [0]*len(neg))
    
    skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
    best_mcc = -1
    
    for fold, (train_idx, val_idx) in enumerate(skf.split(np.zeros(len(y)), y), 1):
        train_loader = DataLoader(TensorDataset(X_tensor[train_idx], torch.tensor(y[train_idx], dtype=torch.float32)), batch_size=64, shuffle=True)
        model = CNN_BiLSTM()
        optimizer = optim.AdamW(model.parameters(), lr=1e-3)
        criterion = nn.BCELoss()
        
        model.train()
        for epoch in range(20):
            for bx, by in train_loader:
                optimizer.zero_grad()
                criterion(model(bx), by).backward()
                optimizer.step()
                
        model.eval()
        with torch.no_grad():
            preds = model(X_tensor[val_idx]).numpy()
        
        mcc = matthews_corrcoef(y[val_idx], (preds >= 0.5).astype(int))
        print(f"Fold {fold}: MCC = {mcc:.4f} | AUC = {roc_auc_score(y[val_idx], preds):.4f}")
        
        if mcc > best_mcc:
            best_mcc = mcc
            torch.save(model.state_dict(), Path(output_dir) / "best_model.pt")
            print(f" -> Saved new best model (MCC: {mcc:.4f})")

if __name__ == '__main__':
    parser = argparse.ArgumentParser()
    parser.add_argument("--pos", required=True)

    parser.add_argument("--out-dir", default="output")
    args = parser.parse_args()
    train_pytorch(args.pos, args.neg, args.out_dir)

import argparse
import joblib
from pathlib import Path
import lightgbm as lgb
import numpy as np
from sklearn.model_selection import StratifiedKFold
from sklearn.metrics import precision_score, recall_score, matthews_corrcoef, roc_auc_score
from dataset_builder import prepare_dataset

def train_lightgbm(data_path, output_dir="output", k_size=3):
    exp_folder = Path(f"{output_dir}")
    exp_folder.mkdir(parents=True, exist_ok=True)

    # 1. Prepare Data
    df = prepare_dataset(data_path, k_size=k_size)
    X = df.drop(columns=['label'])
    y = df['label']
    
    print(f"Dataset prepared: {X.shape[0]} samples, {X.shape[1]} features.")

    # 2. Setup Cross-Validation
    skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
    
    params = {
        'objective': 'binary',
        'metric': 'auc',
        'learning_rate': 0.05,
        'max_depth': 6,
        'num_leaves': 31,
        'verbose': -1,
        'random_state': 42
    }

    best_mcc = -1
    fold_metrics = []

    print("\nStarting 5-Fold Cross Validation...")
    for fold, (train_idx, val_idx) in enumerate(skf.split(X, y), 1):
        X_train, y_train = X.iloc[train_idx], y.iloc[train_idx]
        X_val, y_val = X.iloc[val_idx], y.iloc[val_idx]
        
        train_data = lgb.Dataset(X_train, label=y_train)
        val_data = lgb.Dataset(X_val, label=y_val, reference=train_data)
        
        model = lgb.train(
            params, 
            train_data, 
            num_boost_round=300,
            valid_sets=[val_data],
            callbacks=[lgb.early_stopping(stopping_rounds=30, verbose=False)]
        )
        
        # Evaluate Fold
        val_preds_proba = model.predict(X_val)
        val_preds = (val_preds_proba >= 0.5).astype(int)
        
        precision = precision_score(y_val, val_preds, zero_division=0)
        recall = recall_score(y_val, val_preds, zero_division=0)
        mcc = matthews_corrcoef(y_val, val_preds)
        auc = roc_auc_score(y_val, val_preds_proba)
        
        print(f"Fold {fold}: AUC = {auc:.4f} | MCC = {mcc:.4f} | P = {precision:.4f} | R = {recall:.4f}")
        fold_metrics.append({'auc': auc, 'mcc': mcc, 'precision': precision, 'recall': recall})
        
        # Save best model based on MCC
        if mcc > best_mcc:
            best_mcc = mcc
            joblib.dump(model, exp_folder / "best_model.pkl")
            # We also save the feature columns to ensure alignment during inference
            joblib.dump(list(X.columns), exp_folder / "feature_columns.pkl")
            print(f"  -> New best model saved (MCC: {mcc:.4f})")

    # Log overall metrics
    mean_mcc = np.mean([m['mcc'] for m in fold_metrics])
    mean_auc = np.mean([m['auc'] for m in fold_metrics])
    print(f"\nTraining Complete. Average AUC: {mean_auc:.4f} | Average MCC: {mean_mcc:.4f}")

    with open(exp_folder / "log.txt", "w") as f:
        f.write(f"5-Fold Cross Validation Results:\n")
        f.write(f"Mean AUC: {mean_auc:.4f}\n")
        f.write(f"Mean MCC: {mean_mcc:.4f}\n")
        
if __name__ == '__main__':

import itertools
import random

import numpy as np


def reader(filep):
    with open(filep, "r") as f:
        data = f.readlines()
    data = [da.strip() for da in data]
    return data


def get_list_kmer(kmer, string="TGCA"):
    string = list(string)
    lkmer = ["".join(p) for p in itertools.product(string, repeat=kmer)]
    lkmer.sort()
    return lkmer


def protein2num(protein, elements):
    kmer = len(elements[0])
    edict = dict()
    for i, ele in enumerate(elements):
        edict[ele] = i
    token = list()
    for i in range(len(protein)-kmer+1):
        try:
            token.append(edict[protein[i:i+kmer]])
        except KeyError:
            token.append(0)
            print("Key error : ", protein[i:i+kmer], i)
    return token


def neggen(protein, num_part=20, keep=8, max_class=4):
    length = len(protein)
    # get part
    part_len = length // num_part
    if part_len * num_part < length:
        num_part += 1

    iterator = np.arange(num_part)
    keep_parts = random.sample(list(iterator), k=keep)

    outpro = list()
    for it in iterator:
        start = it * part_len
        pro_part = protein[start:start + part_len]
        if it in keep_parts:
            outpro.extend(pro_part)
        else:
            pro_part = random.choices(np.arange(max_class), k=len(pro_part))
            outpro.extend(pro_part)
    return outpro
